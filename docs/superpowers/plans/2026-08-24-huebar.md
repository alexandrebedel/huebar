```swift
import Foundation

public struct HueConfig: Sendable, Equatable {
    public var bridgeIP: String
    public var appKey: String

    public init(bridgeIP: String, appKey: String) {
        self.bridgeIP = bridgeIP
        self.appKey = appKey
    }

    /// Lit un fichier env de la forme `export KEY=VALUE` (quotes et commentaires tolérés).
    public static func load(path: String? = nil) throws -> HueConfig {
        let resolved = path ?? ("~/.config/hue/credentials.env" as NSString).expandingTildeInPath
        guard let content = try? String(contentsOfFile: resolved, encoding: .utf8) else {
            throw HueError.missingCredentials(resolved)
        }
        var values: [String: String] = [:]
        for rawLine in content.split(separator: "\n") {
            var line = rawLine.trimmingCharacters(in: .whitespaces)
            guard !line.isEmpty, !line.hasPrefix("#") else { continue }
            if line.hasPrefix("export ") { line = String(line.dropFirst("export ".count)) }
            guard let eq = line.firstIndex(of: "=") else { continue }
            let key = String(line[..<eq]).trimmingCharacters(in: .whitespaces)
            let value = String(line[line.index(after: eq)...])
                .trimmingCharacters(in: CharacterSet(charactersIn: " \t\"'"))
            values[key] = value
        }
        guard let bridge = values["HUE_BRIDGE"], let appKey = values["HUE_KEY"],
              !bridge.isEmpty, !appKey.isEmpty else {
            throw HueError.missingCredentials(resolved)
        }
        return HueConfig(bridgeIP: bridge, appKey: appKey)
    }
}
```

```swift
import Foundation

/// Enveloppe générique des réponses CLIP v2 : {"errors":[...],"data":[...]}
public struct HueResponse<T: Decodable>: Decodable {
    public var data: [T]
}

public struct Metadata: Codable, Sendable, Equatable {
    public var name: String
    public init(name: String) { self.name = name }
}

public struct OnState: Codable, Sendable, Equatable {
    public var on: Bool
    public init(on: Bool) { self.on = on }
}

public struct Dimming: Codable, Sendable, Equatable {
    public var brightness: Double
    public init(brightness: Double) { self.brightness = brightness }

    private enum CodingKeys: String, CodingKey { case brightness }
}

public struct XY: Codable, Sendable, Equatable {
    public var x: Double
    public var y: Double
    public init(x: Double, y: Double) { self.x = x; self.y = y }
}

public struct ColorState: Codable, Sendable, Equatable {
    public var xy: XY
    public init(xy: XY) { self.xy = xy }

    private enum CodingKeys: String, CodingKey { case xy }
}

public struct ResourceRef: Codable, Sendable, Equatable {
    public var rid: String
    public var rtype: String
    public init(rid: String, rtype: String) { self.rid = rid; self.rtype = rtype }
}

public struct Light: Codable, Identifiable, Sendable, Equatable {
    public var id: String
    public var metadata: Metadata
    public var on: OnState
    public var dimming: Dimming?
    public var color: ColorState?
    public var owner: ResourceRef
}

public struct Room: Codable, Identifiable, Sendable, Equatable {
    public var id: String
    public var metadata: Metadata
    public var children: [ResourceRef]
    public var services: [ResourceRef]
}

public struct HueScene: Codable, Identifiable, Sendable, Equatable {
    public var id: String
    public var metadata: Metadata
    public var group: ResourceRef
}

public struct GroupedLight: Codable, Identifiable, Sendable, Equatable {
    public var id: String
    public var owner: ResourceRef
    public var on: OnState?
    public var dimming: Dimming?
}

/// Corps des PUT sur light et grouped_light (seuls les champs non-nil sont envoyés).
public struct LightUpdate: Encodable, Sendable, Equatable {
    public var on: OnState?
    public var dimming: Dimming?
    public var color: ColorState?

    public init(on: OnState? = nil, dimming: Dimming? = nil, color: ColorState? = nil) {
        self.on = on
        self.dimming = dimming
        self.color = color
    }

    public func encode(to encoder: Encoder) throws {
        var container = encoder.container(keyedBy: CodingKeys.self)
        try container.encodeIfPresent(on, forKey: .on)
        try container.encodeIfPresent(dimming, forKey: .dimming)
        try container.encodeIfPresent(color, forKey: .color)
    }

    private enum CodingKeys: String, CodingKey { case on, dimming, color }
}
```

Note : les `CodingKeys` explicites sur `Dimming`/`ColorState` évitent d'échouer sur les champs annexes du bridge (`min_dim_level`, `gamut_type`) — le décodage ignore déjà les clés inconnues, ces enums documentent juste le sous-ensemble retenu.

`Sources/HueKit/HueColor.swift` — formule officielle Philips (gamma sRGB + matrice Wide RGB D65) :

```swift
import Foundation

public enum HueColor {
    /// RGB (0...1) -> point xy CIE 1931, formule Philips Hue.
    public static func xy(red: Double, green: Double, blue: Double) -> XY {
        func gamma(_ c: Double) -> Double {
            c > 0.04045 ? pow((c + 0.055) / 1.055, 2.4) : c / 12.92
        }
        let r = gamma(red), g = gamma(green), b = gamma(blue)
        let X = r * 0.664511 + g * 0.154324 + b * 0.162028
        let Y = r * 0.283881 + g * 0.668433 + b * 0.047685
        let Z = r * 0.000088 + g * 0.072310 + b * 0.986039
        let sum = X + Y + Z
        guard sum > 0 else { return XY(x: 0.3127, y: 0.3290) } // blanc D65
        return XY(x: X / sum, y: Y / sum)
    }

    /// Point xy CIE -> RGB (0...1) pour l'affichage (pastilles couleur).
    public static func rgb(from xy: XY, brightness: Double = 1) -> (red: Double, green: Double, blue: Double) {
        guard xy.y > 0 else { return (0, 0, 0) }
        let Y = brightness
        let X = (Y / xy.y) * xy.x
        let Z = (Y / xy.y) * (1 - xy.x - xy.y)

        var r = X * 1.656492 - Y * 0.354851 - Z * 0.255038
        var g = -X * 0.707196 + Y * 1.655397 + Z * 0.036152
        var b = X * 0.051713 - Y * 0.121364 + Z * 1.011530

        let maxComponent = max(r, g, b)
        if maxComponent > 1 { r /= maxComponent; g /= maxComponent; b /= maxComponent }

        func reverseGamma(_ c: Double) -> Double {
            let clamped = max(0, c)
            return clamped <= 0.0031308 ? 12.92 * clamped : 1.055 * pow(clamped, 1 / 2.4) - 0.055
        }
        return (min(1, reverseGamma(r)), min(1, reverseGamma(g)), min(1, reverseGamma(b)))
    }
}
```

```swift
import Foundation

final class MockURLProtocol: URLProtocol {
    nonisolated(unsafe) static var handler: (@Sendable (URLRequest) throws -> (HTTPURLResponse, Data))?

    override class func canInit(with request: URLRequest) -> Bool { true }
    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }

    override func startLoading() {
        guard let handler = Self.handler else {
            fatalError("MockURLProtocol.handler not set")
        }
        do {
            let (response, data) = try handler(request)
            client?.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
            client?.urlProtocol(self, didLoad: data)
            client?.urlProtocolDidFinishLoading(self)
        } catch {
            client?.urlProtocol(self, didFailWithError: error)
        }
    }

    override func stopLoading() {}
}

extension URLRequest {
    /// URLSession transforme httpBody en stream côté URLProtocol : on relit le stream.
    var bodyData: Data? {
        if let body = httpBody { return body }
        guard let stream = httpBodyStream else { return nil }
        stream.open()
        defer { stream.close() }
        var data = Data()
        let bufferSize = 4096
        let buffer = UnsafeMutablePointer<UInt8>.allocate(capacity: bufferSize)
        defer { buffer.deallocate() }
        while stream.hasBytesAvailable {
            let read = stream.read(buffer, maxLength: bufferSize)
            guard read > 0 else { break }
            data.append(buffer, count: read)
        }
        return data
    }
}
```

```swift
final class BridgeTrustDelegate: NSObject, URLSessionDelegate, @unchecked Sendable {

    init(host: String) { self.host = host }

        _ session: URLSession,
        didReceive challenge: URLAuthenticationChallenge,
        completionHandler: @escaping @Sendable (URLSession.AuthChallengeDisposition, URLCredential?) -> Void
    ) {
        guard challenge.protectionSpace.authenticationMethod == NSURLAuthenticationMethodServerTrust,
              challenge.protectionSpace.host == host,
              let trust = challenge.protectionSpace.serverTrust else {
            completionHandler(.performDefaultHandling, nil)
            return
        }
        completionHandler(.useCredential, URLCredential(trust: trust))
    }
}

public final class HueClient: HueControlling, Sendable {
    private let config: HueConfig
    private let session: URLSession

    public init(config: HueConfig, sessionConfiguration: URLSessionConfiguration = .ephemeral) {
        self.config = config
        self.session = URLSession(
            configuration: sessionConfiguration,
            delegate: BridgeTrustDelegate(host: config.bridgeIP),
            delegateQueue: nil
        )
    }

    // MARK: - HueControlling

    public func fetchLights() async throws -> [Light] { try await list("light") }
    public func fetchRooms() async throws -> [Room] { try await list("room") }
    public func fetchScenes() async throws -> [HueScene] { try await list("scene") }
    public func fetchGroupedLights() async throws -> [GroupedLight] { try await list("grouped_light") }

    public func updateLight(id: String, _ update: LightUpdate) async throws {
        _ = try await send("PUT", "/clip/v2/resource/light/\(id)", body: try encode(update))
    }

    public func updateGroupedLight(id: String, _ update: LightUpdate) async throws {
        _ = try await send("PUT", "/clip/v2/resource/grouped_light/\(id)", body: try encode(update))
    }

    public func recallScene(id: String) async throws {
        struct Recall: Encodable { let action = "active" }
        struct Body: Encodable { let recall = Recall() }
        _ = try await send("PUT", "/clip/v2/resource/scene/\(id)", body: try encode(Body()))
    }

    // MARK: - Plomberie

    private func list<T: Decodable>(_ resource: String) async throws -> [T] {
        let data = try await send("GET", "/clip/v2/resource/\(resource)", body: nil)
        return try JSONDecoder().decode(HueResponse<T>.self, from: data).data
    }

    private func encode(_ value: some Encodable) throws -> Data {
        let encoder = JSONEncoder()
        encoder.outputFormatting = .sortedKeys
        return try encoder.encode(value)
    }

    private func send(_ method: String, _ path: String, body: Data?) async throws -> Data {
        guard let url = URL(string: "https://\(config.bridgeIP)\(path)") else {
            throw HueError.invalidResponse
        }
        var request = URLRequest(url: url, timeoutInterval: 2)
        request.httpMethod = method
        request.httpBody = body
        request.setValue(config.appKey, forHTTPHeaderField: "hue-application-key")
        if body != nil {
            request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        }
        let (data, response) = try await session.data(for: request)
        guard let http = response as? HTTPURLResponse else { throw HueError.invalidResponse }
        guard (200..<300).contains(http.statusCode) else { throw HueError.badStatus(http.statusCode) }
        return data
    }
}
```

```swift
import Foundation
import Observation

public struct RoomModel: Identifiable, Sendable, Equatable {
    public let id: String
    public let name: String
    public let groupedLightID: String?
    public var isOn: Bool
    public var brightness: Double
    public var scenes: [HueScene]
    public var lights: [Light]
}

public struct AllZone: Sendable, Equatable {
    public let groupedLightID: String
    public var isOn: Bool
    public var scenes: [HueScene]
}

@MainActor
@Observable
public final class HueState {
    public enum Status: Equatable {
        case loading
        case ready
        case error(String)
    }

    public private(set) var status: Status = .loading
    public var all: AllZone?
    public var rooms: [RoomModel] = []

    private let client: (any HueControlling)?
    private let configErrorMessage: String?

    public init(client: (any HueControlling)?, errorMessage: String? = nil) {
        self.client = client
        self.configErrorMessage = errorMessage
    }

    /// Assemble l'arbre pièces -> lampes/scènes à partir des ressources brutes du bridge.
    /// Mapping : room.children contient des device ids ; light.owner.rid est le device id.
    static func makeRooms(
        lights: [Light], rooms: [Room], scenes: [HueScene], grouped: [GroupedLight]
    ) -> (all: AllZone?, rooms: [RoomModel]) {
        let groupedByOwner = Dictionary(uniqueKeysWithValues: grouped.map { ($0.owner.rid, $0) })

        let roomModels = rooms.map { room -> RoomModel in
            let deviceIDs = Set(room.children.map(\.rid))
            let groupedLight = groupedByOwner[room.id]
            return RoomModel(
                id: room.id,
                name: room.metadata.name,
                groupedLightID: groupedLight?.id,
                isOn: groupedLight?.on?.on ?? false,
                brightness: groupedLight?.dimming?.brightness ?? 100,
                scenes: scenes.filter { $0.group.rid == room.id },
                lights: lights.filter { deviceIDs.contains($0.owner.rid) }
            )
        }
        .sorted { $0.name.localizedStandardCompare($1.name) == .orderedAscending }

        let bridgeHome = grouped.first { $0.owner.rtype == "bridge_home" }
        let allZone = bridgeHome.map {
            AllZone(
                groupedLightID: $0.id,
                isOn: $0.on?.on ?? false,
                scenes: scenes.filter { $0.group.rtype == "bridge_home" }
            )
        }
        return (allZone, roomModels)
    }
}
```

```swift
    // MARK: - Chargement

    public func refresh() async {
        guard let client else {
            status = .error(configErrorMessage ?? "Non configuré")
            return
        }
        do {
            async let lights = client.fetchLights()
            async let rooms = client.fetchRooms()
            async let scenes = client.fetchScenes()
            async let grouped = client.fetchGroupedLights()
            let built = Self.makeRooms(
                lights: try await lights, rooms: try await rooms,
                scenes: try await scenes, grouped: try await grouped
            )
            self.all = built.all
            self.rooms = built.rooms
            self.status = .ready
        } catch {
            status = .error("Bridge injoignable")
        }
    }

    // MARK: - Actions (optimistes : mutation locale immédiate, resync si échec)

    public func toggleAll() async {
        guard var zone = all else { return }
        zone.isOn.toggle()
        all = zone
        await perform { [id = zone.groupedLightID, on = zone.isOn] client in
            try await client.updateGroupedLight(id: id, LightUpdate(on: OnState(on: on)))
        }
    }

    public func toggleRoom(id: String) async {
        guard let index = rooms.firstIndex(where: { $0.id == id }),
              let groupedID = rooms[index].groupedLightID else { return }
        rooms[index].isOn.toggle()
        await perform { [on = rooms[index].isOn] client in
            try await client.updateGroupedLight(id: groupedID, LightUpdate(on: OnState(on: on)))
        }
    }

    public func setRoomBrightness(id: String, _ brightness: Double) async {
        guard let index = rooms.firstIndex(where: { $0.id == id }),
              let groupedID = rooms[index].groupedLightID else { return }
        rooms[index].brightness = brightness
        rooms[index].isOn = true
        await perform { client in
            try await client.updateGroupedLight(
                id: groupedID,
                LightUpdate(on: OnState(on: true), dimming: Dimming(brightness: brightness))
            )
        }
    }

    public func recallScene(id: String) async {
        await perform { client in
            try await client.recallScene(id: id)
        }
        await refresh() // une scène change lampes et couleurs : on resynchronise tout
    }

    public func toggleLight(roomID: String, lightID: String) async {
        guard let (roomIndex, lightIndex) = lightIndices(roomID: roomID, lightID: lightID) else { return }
        rooms[roomIndex].lights[lightIndex].on.on.toggle()
        await perform { [on = rooms[roomIndex].lights[lightIndex].on.on] client in
            try await client.updateLight(id: lightID, LightUpdate(on: OnState(on: on)))
        }
    }

    public func setLightBrightness(roomID: String, lightID: String, _ brightness: Double) async {
        guard let (roomIndex, lightIndex) = lightIndices(roomID: roomID, lightID: lightID) else { return }
        rooms[roomIndex].lights[lightIndex].dimming = Dimming(brightness: brightness)
        rooms[roomIndex].lights[lightIndex].on = OnState(on: true)
        await perform { client in
            try await client.updateLight(
                id: lightID,
                LightUpdate(on: OnState(on: true), dimming: Dimming(brightness: brightness))
            )
        }
    }

    public func setLightColor(roomID: String, lightID: String, red: Double, green: Double, blue: Double) async {
        guard let (roomIndex, lightIndex) = lightIndices(roomID: roomID, lightID: lightID) else { return }
        let xy = HueColor.xy(red: red, green: green, blue: blue)
        rooms[roomIndex].lights[lightIndex].color = ColorState(xy: xy)
        rooms[roomIndex].lights[lightIndex].on = OnState(on: true)
        await perform { client in
            try await client.updateLight(
                id: lightID,
                LightUpdate(on: OnState(on: true), color: ColorState(xy: xy))
            )
        }
    }

    // MARK: - Helpers

    private func lightIndices(roomID: String, lightID: String) -> (Int, Int)? {
        guard let roomIndex = rooms.firstIndex(where: { $0.id == roomID }),
              let lightIndex = rooms[roomIndex].lights.firstIndex(where: { $0.id == lightID })
        else { return nil }
        return (roomIndex, lightIndex)
    }

    /// Exécute une commande ; en cas d'échec, resynchronise tout (refresh pose le bon status).
    private func perform(_ operation: (any HueControlling) async throws -> Void) async {
        guard let client else { return }
        do {
            try await operation(client)
            if status != .ready { status = .ready }
        } catch {
            await refresh()
        }
    }
```

```swift
struct LightRowView: View {
    let state: HueState
    let roomID: String
    let light: Light
    @State private var brightness: Double = 100
    @State private var color: Color = .white
    @State private var colorTask: Task<Void, Never>?

            HStack(spacing: 8) {
                Toggle("", isOn: Binding(
                    get: { light.on.on },

                ))

                .controlSize(.mini)
                .labelsHidden()

                Text(light.metadata.name)
                    .font(.callout)
                    .lineLimit(1)

                Spacer()

                if light.color != nil {
                    ColorPicker("", selection: $color, supportsOpacity: false)
                        .labelsHidden()
                        .controlSize(.small)
                        .onChange(of: color) { _, newValue in
                            scheduleColorSend(newValue)
                        }
                }
            }
            if light.dimming != nil {
                Slider(value: $brightness, in: 1...100) { editing in
                    if !editing {
                        Task { await state.setLightBrightness(roomID: roomID, lightID: light.id, brightness) }
                    }
                }
                .controlSize(.mini)
            }
        }
        .padding(.leading, 4)
        .onAppear {
            brightness = light.dimming?.brightness ?? 100
            if let xy = light.color?.xy {
                let rgb = HueColor.rgb(from: xy)
                color = Color(red: rgb.red, green: rgb.green, blue: rgb.blue)
            }
        }
    }

    /// Le ColorPicker émet en continu pendant le drag : on n'envoie que la dernière valeur (250 ms).
    private func scheduleColorSend(_ newColor: Color) {
        colorTask?.cancel()
        colorTask = Task {
            try? await Task.sleep(for: .milliseconds(250))
            guard !Task.isCancelled else { return }
            guard let rgb = NSColor(newColor).usingColorSpace(.sRGB) else { return }
            await state.setLightColor(
                roomID: roomID, lightID: light.id,
                red: rgb.redComponent, green: rgb.greenComponent, blue: rgb.blueComponent
            )
        }
    }
}
```

```swift
import AppKit
import Carbon.HIToolbox

/// Hotkey global via l'API Carbon (aucune permission d'accessibilité requise).
final class HotKeyCenter {
    static var action: (() -> Void)?
    private var hotKeyRef: EventHotKeyRef?

    /// Enregistre ⌥H.
    func register() {
        var eventType = EventTypeSpec(
            eventClass: OSType(kEventClassKeyboard),
            eventKind: UInt32(kEventHotKeyPressed)
        )
        InstallEventHandler(GetApplicationEventTarget(), { _, _, _ in
            DispatchQueue.main.async { HotKeyCenter.action?() }
            return noErr
        }, 1, &eventType, nil, nil)

        let hotKeyID = EventHotKeyID(signature: OSType(0x48554241), id: 1) // "HUBA"
        RegisterEventHotKey(
            UInt32(kVK_ANSI_H), UInt32(optionKey), hotKeyID,
            GetApplicationEventTarget(), 0, &hotKeyRef
        )
    }
}
```

```swift
import AppKit
import SwiftUI
import HueKit

final class AppDelegate: NSObject, NSApplicationDelegate {
    private var statusItem: NSStatusItem!
    private let popover = NSPopover()
    private let hotKeys = HotKeyCenter()
    private var state: HueState!

    func applicationDidFinishLaunching(_ notification: Notification) {
        do {
            state = HueState(client: HueClient(config: try HueConfig.load()))
        } catch {
            state = HueState(
                client: nil,
                errorMessage: "Credentials introuvables (~/.config/hue/credentials.env)"
            )
        }

        popover.behavior = .transient
        popover.contentViewController = NSHostingController(rootView: PanelView(state: state))

        statusItem = NSStatusBar.system.statusItem(withLength: NSStatusItem.squareLength)
        if let button = statusItem.button {
            button.image = NSImage(systemSymbolName: "lightbulb.fill", accessibilityDescription: "HueBar")
            button.target = self
            button.action = #selector(togglePopover)
        }

        HotKeyCenter.action = { [weak self] in self?.togglePopover() }
        hotKeys.register()

        registerLoginItemOnFirstLaunch()
    }

    @objc private func togglePopover() {
        if popover.isShown {
            popover.performClose(nil)
        } else if let button = statusItem.button {
            Task { await state.refresh() }
            popover.show(relativeTo: button.bounds, of: button, preferredEdge: .minY)
            popover.contentViewController?.view.window?.makeKey()
            NSApp.activate()
        }
    }

    /// « Ouvrir au login » actif par défaut, mais on respecte un choix utilisateur ultérieur.
    private func registerLoginItemOnFirstLaunch() {
        let key = "didConfigureLoginItem"
        guard !UserDefaults.standard.bool(forKey: key) else { return }
        LoginItem.set(true)
        UserDefaults.standard.set(true, forKey: key)
    }
}
```
